import Demo from '../../components/Demo.astro'
import Pagination from '@oneonefour/rkui/components/pagination/Pagination.vue'

Give it the `current` page, the `total` and an `href` function that returns a page's link. It
links the first and last pages, the current one and its neighbours, with previous and next.

<Demo label="Pagination" stack>
  <Pagination current={2} total={9} href={(page) => `#page-${page}`}>
    <p>Page 2.</p>
  </Pagination>
</Demo>

## On the first page

There is no previous link on the first page, and no next link on the last.

<Demo label="Pagination on the first page" stack>
  <Pagination current={1} total={3} href={(page) => `#page-${page}`}>
    <p>Page 1: no previous link.</p>
  </Pagination>
</Demo>

## Scroll

Links that don't fit scroll, with dots at the strip's ends; `dots="pill"` puts them in pills (see
[Scroll](../scroll/)).

<Demo label="Narrow pagination that scrolls" stack>
  <Pagination
    current={10}
    total={20}
    href={(page) => `#page-${page}`}
    overflow="scroll"
    style="max-inline-size: 16rem"
  >
    <p>Page 10 of 20.</p>
  </Pagination>
</Demo>
